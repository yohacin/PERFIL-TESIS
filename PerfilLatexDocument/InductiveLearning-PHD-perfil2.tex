% --- Chapter numbering (Capítulo N) -----------------------------------
% \Capitulo{Título} prints the centered "Capítulo N. Título" heading,
% adds it to the TOC, and rebinds \thesubsection to N.M so that every
% \subsection / \subsubsection inside the chapter numbers and TOC-lists
% itself automatically (no manual "2.3." typing, no \addcontentsline).
\newcounter{capitulo}
\newcommand{\Capitulo}[1]{%
	\stepcounter{capitulo}%
	% Avanza el contador de sección para que los destinos internos del PDF sean
	% únicos en cada capítulo; no altera la numeración visible, que se toma del
	% contador capitulo.
	\refstepcounter{section}%
	\setcounter{subsection}{0}%
	\renewcommand{\thesubsection}{\arabic{capitulo}.\arabic{subsection}}%
	\clearpage
	\begin{center}
		{\normalsize\bfseries Capítulo \arabic{capitulo}.\\ #1}
	\end{center}
	\addcontentsline{toc}{section}{Capítulo \arabic{capitulo}. #1}%
}
